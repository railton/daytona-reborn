cmake_minimum_required(VERSION 3.20)
project(producer_ip LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(RUNTIME_BUILD "" CACHE PATH "Existing compatible desktop build with m2recomp and runtime libraries")
if(NOT IS_DIRECTORY "${RUNTIME_BUILD}")
  message(FATAL_ERROR "Set RUNTIME_BUILD to the existing desktop build")
endif()
find_program(M2RECOMP_EXECUTABLE NAMES m2recomp
  PATHS "${RUNTIME_BUILD}/scene-preview" "${RUNTIME_BUILD}/Release" "${RUNTIME_BUILD}"
  NO_DEFAULT_PATH REQUIRED)
foreach(lib runtime trace i960 softfloat)
  find_library(${lib}_LIBRARY NAMES ${lib} PATHS "${RUNTIME_BUILD}/Release" "${RUNTIME_BUILD}"
    NO_DEFAULT_PATH REQUIRED)
endforeach()
set(program "${CMAKE_CURRENT_BINARY_DIR}/synthetic.bin")
add_custom_command(OUTPUT "${program}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/make_program.py" "${program}"
  DEPENDS make_program.py VERBATIM)
add_custom_target(producer_ip_program DEPENDS "${program}")
foreach(mode exact block exact_hook block_hook block_late)
  set(output "${CMAKE_CURRENT_BINARY_DIR}/${mode}")
  set(flags)
  set(block_mode 0)
  set(hook_mode 0)
  set(hook_manifest)
  if(mode MATCHES "^block")
    set(flags --fast_inaccuracy)
    set(block_mode 1)
  endif()
  if(mode MATCHES "_hook$")
    set(hook_mode 1)
    set(hook_manifest "${CMAKE_CURRENT_SOURCE_DIR}/first_store.hooks")
  elseif(mode STREQUAL block_late)
    set(hook_mode 2)
    set(hook_manifest "${CMAKE_CURRENT_SOURCE_DIR}/second_store.hooks")
  endif()
  if(hook_manifest)
    list(APPEND flags --hooks "${hook_manifest}")
  endif()
  add_custom_command(OUTPUT "${output}/chunk_000.cpp" "${output}/gen_table.cpp"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${output}"
    COMMAND "${M2RECOMP_EXECUTABLE}" "${program}" "${output}" ${flags}
    DEPENDS producer_ip_program "${program}" "${M2RECOMP_EXECUTABLE}" ${hook_manifest} VERBATIM)
  add_executable(${mode} main.cpp "${output}/chunk_000.cpp" "${output}/gen_table.cpp")
  target_include_directories(${mode} PRIVATE "${PROJECT_ROOT}/src" "${PROJECT_ROOT}/extern/softfloat/source/include")
  target_compile_definitions(${mode} PRIVATE PROBE_BLOCK=${block_mode} PROBE_HOOK=${hook_mode})
  target_precompile_headers(${mode} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/probe_hook.h")
  target_link_libraries(${mode} PRIVATE ${runtime_LIBRARY} ${trace_LIBRARY} ${i960_LIBRARY} ${softfloat_LIBRARY})
  if(NOT hook_mode)
    add_test(NAME producer_ip_${mode} COMMAND ${mode})
  endif()
  add_test(NAME producer_boundary_${mode} COMMAND ${mode} --boundary)
endforeach()
